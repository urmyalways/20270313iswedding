            function initAndPlayGentleMusic() {

                try {

                    if (typeof Tone === 'undefined') return;

                    if (!synth) {

                        const reverb = new Tone.Reverb({ decay: 4.5, wet: 0.55 }).toDestination();

                        synth = new Tone.PolySynth(Tone.Synth, {

                            oscillator: { type: 'sine' },

                            envelope: {

                                attack: 0.4,

                                decay: 1.2,

                                sustain: 0.4,

                                release: 2.5

                            },

                            volume: -14

                        }).connect(reverb);


                        const melodyNotes = [

                            ['C4', 'E4', 'G4', 'D5'],

                            ['B3', 'D4', 'G4', 'D5'],

                            ['A3', 'C4', 'E4', 'G4'],

                            ['F3', 'A3', 'C4', 'E4']

                        ];

                        let step = 0;


                        musicLoop = new Tone.Loop((time) => {

                            const chord = melodyNotes[step % melodyNotes.length];

                            chord.forEach((note, idx) => {

                                synth.triggerAttackRelease(note, '2n', time + idx * 0.45);

                            });

                            step++;

                        }, '2m');

                    }


                    Tone.start().then(() => {

                        Tone.Transport.bpm.value = 68;

                        musicLoop.start(0);

                        Tone.Transport.start();

                        isMusicPlaying = true;

                        musicBtn.classList.remove('music-paused');

                    });

                } catch (e) {

                    console.warn('Audio playback notice:', e);

                }

            }


            musicBtn.addEventListener('click', () => {

                if (!synth) {

                    initAndPlayGentleMusic();

                    return;

                }

                if (isMusicPlaying) {

                    Tone.Transport.pause();

                    isMusicPlaying = false;

                    musicBtn.classList.add('music-paused');

                } else {

                    Tone.Transport.start();

                    isMusicPlaying = true;

                    musicBtn.classList.remove('music-paused');

                }

            });

            function updateCountdown() {

                const now = new Date().getTime();

                const distance = targetDate - now;


                if (distance <= 0) {

                    elDays.textContent = '000';

                    elHours.textContent = '00';

                    elMinutes.textContent = '00';

                    elSeconds.textContent = '00';

                    return;

                }


                const days = Math.floor(distance / (1000 * 60 * 60 * 24));

                const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

                const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));

                const seconds = Math.floor((distance % (1000 * 60)) / 1000);


                elDays.textContent = String(days).padStart(3, '0');

                elHours.textContent = String(hours).padStart(2, '0');

                elMinutes.textContent = String(minutes).padStart(2, '0');

                elSeconds.textContent = String(seconds).padStart(2, '0');

            }

            setInterval(updateCountdown, 1000);

            function triggerScrollReveal() {

                const reveals = document.querySelectorAll('.reveal-up');

                const observer = new IntersectionObserver((entries) => {

                    entries.forEach(entry => {

                        if (entry.isIntersecting) {

                            entry.target.classList.add('active');

                        }

                    });

                }, { threshold: 0.12 });


                reveals.forEach(el => observer.observe(el));

            }

        });
